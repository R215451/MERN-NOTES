# ___>>> How to Use Mongoose in Node Js ??

---

# Step 1 → Install Mongoose

```bash
npm install mongoose
```

---

# Step 2 → Connect MongoDB with Mongoose

```js
import mongoose from 'mongoose'

const url = 'mongodb://127.0.0.1:27017/';
const database_name = 'school';

const connection = async()=>{

    await mongoose.connect(
        `${url}${database_name}`
    );

    console.log('MongoDB Connected');

    // Create Schema

    const emp_schema =
    mongoose.Schema({

        name:String,

        age:Number,

        grade:String,

        address:String

    });

    // Create Model

    const student_model =
    new mongoose.model(
        'students',
        emp_schema
    );

    // Fetch Data

    const response =
    await student_model.find();

    console.log(response);

}

connection();
```

# ___>>> CRUD Operations Using Mongoose

---

# Complete CRUD API Example

```js
import mongoose from 'mongoose'
import express from 'express'
import student_model from './model_mongoose/studentModel.js';
import { ObjectId } from 'mongodb';
import cors from 'cors'

const app = express()

// Built in Middlewares

app.use(cors())

app.use(express.json())

app.use(express.urlencoded({
    extended:true
}))

const url =
'mongodb://127.0.0.1:27017/';

const database_name =
'school';

// MongoDB Connection

mongoose.connect(
    `${url}${database_name}`
).then(()=>{

    console.log('MongoDB Connected');

})

// Get All Students

app.get('',async(req,res)=>{

    let response =
    await student_model.find()

    res.send(response)

})

// Save Student

app.post('/save',async(req,res)=>{

    const {
        name,
        age,
        grade,
        address
    } = req.body;

    console.log(req.body);

    if (
        !name ||
        !age ||
        !grade ||
        !address
    ){

        res.send({
            result:false
        })

        return false
    }

    let response =
    await student_model.create(req.body)

    res.send(response)

})

// Update Student

app.put('/update/:id',async(req,res)=>{

    const id = req.params.id;

    console.log(req.body);

    let response =
    await student_model.updateOne(
        {
            _id:new ObjectId(id)
        },
        {
            $set:req.body
        }
    )

    res.send(response)

})

// Delete Student

app.delete('/delete/:id',async(req,res)=>{

    const id = req.params.id;

    let response =
    await student_model.deleteOne({
        _id:new ObjectId(id)
    })

    res.send({
        message:"success",
        result:response
    })

})

app.listen(3200);
```

# ___>>> What is CORS Error ??

CORS stands for:

```txt
Cross Origin Resource Sharing
```

CORS error occurs when:

- frontend and backend run on different ports/domains
- browser blocks request for security reasons

---

# Example

| Frontend | Backend |
|---|---|
| localhost:3000 | localhost:3200 |

Browser blocks API request ❌

---

# Solution

Install CORS package:

```bash
npm install cors
```

Use middleware:

```js
app.use(cors())
```

# Complete File Upload Code

```js
import express from 'express'
import multer from 'multer'

const port = 2700;

const app = express();

const storage = multer.diskStorage({

    destination:(req,file,cb)=>{

        cb(null,'uploads')

    },

    filename:(req,file,cb)=>{

        cb(
            null,
            file.originalname
        )

    }

})

const upload = multer({

    storage:storage

})

app.get('/upload',(req,res)=>{

    res.send(`

        <form
            action='/upload/file'
            method='post'
            enctype='multipart/form-data'
        >

            <input
                type='file'
                name='myfile'
            />

            <button>
                Upload File
            </button>

        </form>

    `)

})

app.post(

    '/upload/file',

    upload.single('myfile'),

    (req,res)=>{

        console.log(req.file);

        res.send({

            message:'PDF Uploaded',

            file:req.file

        })

})

app.listen(port,()=>{

    console.log(
        `server is running at ${port}`
    );

});
```
